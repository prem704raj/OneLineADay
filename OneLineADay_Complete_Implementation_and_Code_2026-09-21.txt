private fun copyZipEntryWithExactLimit(
    input: java.io.InputStream,
    destination: File,
    expectedBytes: Long
) {
    require(
        expectedBytes in
            0..MAX_SINGLE_FILE_BYTES
    ) {
        "Invalid expected attachment size."
    }

    FileOutputStream(
        destination
    ).use { output ->

        val buffer =
            ByteArray(
                DEFAULT_BUFFER_SIZE
            )

        var total =
            0L

        while (true) {
            val read =
                input.read(
                    buffer
                )

            if (read < 0) {
                break
            }

            total +=
                read

            require(
                total <=
                    expectedBytes
            ) {
                "Attachment exceeded declared size."
            }

            output.write(
                buffer,
                0,
                read
            )
        }

        output.fd.sync()

        require(
            total ==
                expectedBytes
        ) {
            "Attachment size validation failed."
        }
    }
}

private suspend fun cleanupNoLongerReferenced(
    previous: JournalEntry?,
    current: JournalEntry?
) {
    if (previous == null) {
        return
    }

    val candidates =
        previous.attachmentPaths() -
            (
                current
                    ?.attachmentPaths()
                    ?: emptySet()
            )

    if (candidates.isEmpty()) {
        return
    }

    val allEntries =
        repository
            .getAllEntriesForBackup()

    val stillReferenced =
        allEntries
            .flatMap {
                it.attachmentPaths()

            .toSet()

    candidates.forEach { path ->
        mediaStorage
            .deleteIfUnreferenced(
                path =

                referencedPaths =
                    stillReferenced
            )
    }
}

package com.onelineaday.dailydiary.ads

import android.app.Activity
import android.content.Context
import com.google.android.gms.ads.MobileAds
import com.google.android.ump.ConsentInformation
import com.google.android.ump.ConsentRequestParameters
import com.google.android.ump.UserMessagingPlatform
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import java.util.concurrent.atomic.AtomicBoolean

object AdsConsentManager {

    private val _canRequestAds =
        MutableStateFlow(false)

    val canRequestAds:
        StateFlow<Boolean> =
        _canRequestAds
            .asStateFlow()

    private val _privacyOptionsRequired =
        MutableStateFlow(false)

    val privacyOptionsRequired:
        StateFlow<Boolean> =
        _privacyOptionsRequired
            .asStateFlow()

    private val mobileAdsInitialized =
        AtomicBoolean(false)

    fun gatherConsent(
        activity: Activity
    ) {
        val consentInformation =
            UserMessagingPlatform
                .getConsentInformation(
                    activity
                )

        val params =
            ConsentRequestParameters
                .Builder()
                .build()

        consentInformation
            .requestConsentInfoUpdate(
                activity,
                params,
                {
                    updateState(
                        consentInformation
                    )

                    UserMessagingPlatform
                        .loadAndShowConsentFormIfRequired(
                            activity
                        ) {
                            _ ->
                            updateState(
                                consentInformation
                            )

                            initializeAdsIfAllowed(
                                activity.applicationContext
                            )
                        }
                },
                {
                    _ ->
                    // Do not guess consent. Use the current UMP state.
                    updateState(
                        consentInformation
                    )

                    initializeAdsIfAllowed(
                        activity.applicationContext
                    )
                }
            )
    }

    private fun updateState(
        info: ConsentInformation
    ) {
        _canRequestAds.value =
            info.canRequestAds()

        _privacyOptionsRequired.value =
            info.privacyOptionsRequirementStatus ==
                ConsentInformation
                    .PrivacyOptionsRequirementStatus
                    .REQUIRED
    }

    private fun initializeAdsIfAllowed(
        context: Context
    ) {
        if (
            !_canRequestAds.value
        ) {
            return
        }

        if (
            mobileAdsInitialized
                .compareAndSet(
                    false,
                    true
                )
        ) {
            MobileAds.initialize(
                context
            ) {
                InterstitialAdManager
                    .loadAd(context)

                RewardedAdManager
                    .loadAd(context)
            }
        }
    }

    fun showPrivacyOptions(
        activity: Activity,
        onComplete: (
            String?
        ) -> Unit = {}
    ) {
        UserMessagingPlatform
            .showPrivacyOptionsForm(
                activity
            ) {
                formError ->

                val info =
                    UserMessagingPlatform
                        .getConsentInformation(
                            activity
                        )

                updateState(info)

                initializeAdsIfAllowed(
                    activity.applicationContext
                )

                onComplete(
                    formError
                        ?.message
                )
            }
    }

    fun resetForDebugOnly(
        context: Context
    ) {
        if (
            com.onelineaday.dailydiary
                .BuildConfig.DEBUG
        ) {
            UserMessagingPlatform
                .getConsentInformation(
                    context
                )
                .reset()

            _canRequestAds.value =
                false

            _privacyOptionsRequired.value =
                false
        }
    }
}

object RewardedAdManager {

    private const val TAG =
        "RewardedAdManager"

    private const val AD_UNIT_ID =
        "ca-app-pub-8204679574020840/8271044708"

    private var rewardedAd:
        RewardedAd? =
        null

    private var loading =
        false

    fun loadAd(
        context: Context
    ) {
        if (
            PremiumManager
                .isPremium
                .value ||
            !AdsConsentManager
                .canRequestAds
                .value ||
            rewardedAd != null ||
            loading
        ) {
            return
        }

        loading =
            true

        RewardedAd.load(
            context,
            AD_UNIT_ID,
            AdRequest
                .Builder()
                .build(),
            object :
                RewardedAdLoadCallback() {

                override fun onAdLoaded(
                    ad: RewardedAd
                ) {
                    rewardedAd =
                        ad

                    loading =
                        false
                }

                override fun onAdFailedToLoad(
                    error: LoadAdError
                ) {
                    Log.w(
                        TAG,
                        error.message
                    )

                    rewardedAd =
                        null

                    loading =
                        false
                }
            }
        )
    }

    fun isAdReady():
        Boolean =
        !PremiumManager
            .isPremium
            .value &&
            AdsConsentManager
                .canRequestAds
                .value &&
            rewardedAd != null

    fun showAd(
        activity: Activity,
        onRewarded: () -> Unit,
        onAdDismissed: () -> Unit,
        onAdNotAvailable: () -> Unit
    ) {
        if (
            PremiumManager
                .isPremium
                .value
        ) {
            onAdNotAvailable()
            return
        }

        if (
            !AdsConsentManager
                .canRequestAds
                .value
        ) {
            /*
             * Export must not be blocked because consent prevents ads.
             * Never punish the user for declining advertising consent.
             */
            onAdNotAvailable()
            return
        }

        val ad =
            rewardedAd

        if (ad == null) {
            onAdNotAvailable()
            loadAd(activity.applicationContext)
            return
        }

        rewardedAd =
            null

        var rewardEarned =
            false

        ad.fullScreenContentCallback =
            object :
                FullScreenContentCallback() {

                override fun onAdDismissedFullScreenContent() {
                    if (
                        rewardEarned
                    ) {
                        onRewarded()
                    }

                    onAdDismissed()

                    loadAd(
                        activity.applicationContext
                    )
                }

                override fun onAdFailedToShowFullScreenContent(
                    adError: AdError
                ) {
                    Log.w(
                        TAG,
                        adError.message
                    )

                    onAdNotAvailable()
                    onAdDismissed()

                    loadAd(
                        activity.applicationContext
                    )
                }
            }

        ad.show(
            activity
        ) {
            rewardEarned =
                true
        }
    }
}

class ReminderManager(
    private val context: Context
) {

    companion object {
        const val CHANNEL_ID =
            "daily_reminder"

        const val NOTIFICATION_ID =
            1001

        const val REQUEST_CODE =
            2001
    }

    init {
        createNotificationChannel()
    }

    private fun createNotificationChannel() {

        if (
            Build.VERSION.SDK_INT <
            Build.VERSION_CODES.O
        ) {
            return
        }

        val manager =
            context.getSystemService(
                NotificationManager::class.java
            )

        val channel =
            NotificationChannel(
                CHANNEL_ID,
                context.getString(
                    R.string.reminder_channel_name
                ),
                NotificationManager
                    .IMPORTANCE_DEFAULT
            ).apply {
                description =
                    context.getString(
                        R.string.reminder_channel_description
                    )

                enableVibration(true)
                setShowBadge(true)
            }

        manager.createNotificationChannel(
            channel
        )
    }

    fun scheduleDailyReminder(
        hour: Int = 20,
        minute: Int = 0
    ) {
        require(hour in 0..23)
        require(minute in 0..59)

        val now =
            ZonedDateTime.now()

        var next =
            now
                .withHour(hour)
                .withMinute(minute)
                .withSecond(0)
                .withNano(0)

        if (
            !next.isAfter(now)
        ) {
            next =
                next.plusDays(1)
        }

        val manager =
            context.getSystemService(
                Context.ALARM_SERVICE
            ) as AlarmManager

        manager.setAndAllowWhileIdle(
            AlarmManager.RTC_WAKEUP,
            next.toInstant()
                .toEpochMilli(),
            reminderPendingIntent()
        )
    }

    fun cancelReminder() {
        val manager =
            context.getSystemService(
                Context.ALARM_SERVICE
            ) as AlarmManager

        manager.cancel(
            reminderPendingIntent()
        )
    }

    private fun reminderPendingIntent():
        PendingIntent {

        val intent =
            Intent(
                context,
                ReminderReceiver::class.java
            )

        return PendingIntent.getBroadcast(
            context,
            REQUEST_CODE,
            intent,
            PendingIntent.FLAG_UPDATE_CURRENT or
                PendingIntent.FLAG_IMMUTABLE
        )
    }
}

class ReminderReceiver :

    ) {

            )

                false
            )
        ) {
            ReminderManager(context)

                    hour =
                        prefs.getInt(
                            "reminder_hour",
                            20
                        ),
                    minute =
                        prefs.getInt(
                            "reminder_minute",
                            0
                        )
                )
        }
    }

    private fun showNotification(
        context: Context
    ) {
        val manager =
            context.getSystemService(
                Context.NOTIFICATION_SERVICE
            ) as NotificationManager

        val openIntent =

                        .CATEGORY_REMINDER

        manager.notify(
            ReminderManager
                .NOTIFICATION_ID,

        )
    }
}

import androidx.core.content.FileProvider
import com.onelineaday.dailydiary.data.JournalEntry
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.File
import java.io.FileOutputStream
import java.time.format.DateTimeFormatter

object PdfExportHelper {

    private const val PAGE_WIDTH =
        595

    private const val PAGE_HEIGHT =
        842

    private const val MARGIN =
        42f

    private const val CONTENT_WIDTH =
        PAGE_WIDTH -
            (MARGIN * 2).toInt()

    suspend fun generateJournalPdf(
        context: Context,
        entries: List<JournalEntry>,
        currentStreak: Int,
        longestStreak: Int,
        totalEntries: Int
    ): Uri? =
        withContext(
            Dispatchers.IO
        ) {
            if (
                entries.isEmpty()
            ) {
                return@withContext null
            }

            val exportDir =
                File(
                    context.cacheDir,
                    "journal_exports"
                ).apply {
                    mkdirs()
                }

            val output =
                File(
                    exportDir,
                    "OneLineADay_" +
                        System.currentTimeMillis() +
                        ".pdf"
                )

            val pdf =
                PdfDocument()

            try {
                val renderer =
                    JournalPdfRenderer(
                        pdf =
                            pdf
                    )

                renderer.drawCover(
                    entries =
                        entries,
                    currentStreak =
                        currentStreak,
                    longestStreak =
                        longestStreak,
                    totalEntries =
                        totalEntries
                )

                entries
                    .sortedByDescending {
                        it.date
                    }
                    .forEach {
                        entry ->
                        renderer.drawEntry(
                            entry
                        )
                    }

                renderer.finish()

                FileOutputStream(
                    output
                ).use {
                    stream ->
                    pdf.writeTo(
                        stream
                    )

                    stream.fd.sync()
                }

                FileProvider.getUriForFile(
                    context,
                    "${context.packageName}.provider",
                    output
                )

            } finally {
                pdf.close()
            }
        }

    private class JournalPdfRenderer(
        private val pdf: PdfDocument
    ) {
        private var pageNumber =
            0

        private var page:
            PdfDocument.Page? =
            null

        private var canvas:
            Canvas? =
            null

        private var y =
            MARGIN

        private val titlePaint =
            TextPaint(
                Paint.ANTI_ALIAS_FLAG
            ).apply {
                color =
                    Color.rgb(
                        40,
                        40,
                        45
                    )

                textSize =
                    25f

                typeface =
                    Typeface.create(
                        Typeface.DEFAULT,
                        Typeface.BOLD
                    )
            }

        private val headingPaint =
            TextPaint(
                Paint.ANTI_ALIAS_FLAG
            ).apply {
                color =
                    Color.rgb(
                        50,
                        50,
                        55
                    )

                textSize =
                    16f

                typeface =
                    Typeface.create(
                        Typeface.DEFAULT,
                        Typeface.BOLD
                    )
            }

        private val bodyPaint =
            TextPaint(
                Paint.ANTI_ALIAS_FLAG
            ).apply {
                color =
                    Color.rgb(
                        55,
                        55,
                        60
                    )

                textSize =
                    13f

                typeface =
                    Typeface.create(
                        Typeface.DEFAULT,
                        Typeface.NORMAL
                    )
            }

        private val mutedPaint =
            TextPaint(
                Paint.ANTI_ALIAS_FLAG
            ).apply {
                color =
                    Color.rgb(
                        105,
                        105,
                        115
                    )

                textSize =
                    10f
            }

        private val linePaint =
            Paint(
                Paint.ANTI_ALIAS_FLAG
            ).apply {
                color =
                    Color.rgb(
                        225,
                        225,
                        230
                    )

                strokeWidth =
                    1f
            }

        fun drawCover(
            entries: List<JournalEntry>,
            currentStreak: Int,
            longestStreak: Int,
            totalEntries: Int
        ) {
            startPage()

            y =
                180f

            drawText(
                text =
                    "One Line A Day",
                paint =
                    titlePaint,
                spacingAfter =
                    14f
            )

            drawText(
                text =
                    "Your personal journal",
                paint =
                    bodyPaint,
                spacingAfter =
                    28f
            )

            val ordered =
                entries.sortedBy {
                    it.date
                }

            val range =
                if (
                    ordered.isNotEmpty()
                ) {
                    val first =
                        ordered.first()
                            .date
                            .format(
                                DateTimeFormatter
                                    .ofPattern(
                                        "MMM yyyy"
                                    )
                            )

                    val last =
                        ordered.last()
                            .date
                            .format(
                                DateTimeFormatter
                                    .ofPattern(
                                        "MMM yyyy"
                                    )
                            )

                    "$first — $last"
                } else {
                    ""
                }

            drawText(
                text =
                    range,
                paint =
                    mutedPaint,
                spacingAfter =
                    28f
            )

            drawText(
                text =
                    "Entries: $totalEntries",
                paint =
                    headingPaint
            )

            drawText(
                text =
                    "Current streak: $currentStreak",

                    bodyPaint
            )

            drawText(
                text =

                paint =
                    bodyPaint
            )

        }

        fun drawEntry(
            entry: JournalEntry
        ) {
            ensurePage()

            ensureSpace(
                155f
            )

            val date =
                entry.date.format(
                    DateTimeFormatter
                        .ofPattern(
                            "MMMM d, yyyy"
                        )
                )

            drawText(
                text =
                    date,
                paint =
                    headingPaint,
                spacingAfter =
                    4f
            )

            drawText(
                text =
                    entry.mood.emoji +
                        " " +
                        entry.mood.name
                            .lowercase()
                            .replaceFirstChar {
                                it.uppercase()
                            },
                paint =
                    mutedPaint,
                spacingAfter =
                    8f
            )

            drawText(
                text =
                    entry.content,
                paint =
                    bodyPaint,
                spacingAfter =
                    8f
            )

            if (
                entry.tags
                    .isNotEmpty()
            ) {
                drawText(
                    text =
                        entry.tags
                            .joinToString(
                                " "
                            ) {
                                "#$it"
                            },
                    paint =
                        mutedPaint,
                    spacingAfter =
                        8f
                )
            }

            if (
                !entry.audioUri
                    .isNullOrBlank()
            ) {
                drawText(
                    text =
                        "Voice note attached",
                    paint =
                        mutedPaint,
                    spacingAfter =
                        6f
                )
            }

            val media =
                (
                    listOfNotNull(
                        entry.photoUri
                    ) +
                        entry.mediaUris
                )

            media.forEach {

                }
            }

            val c =
                canvas
                    ?: return

                20f
            )

                y,
                PAGE_WIDTH -
                    MARGIN,
                y,

            )

            y +=

        }

        private fun drawText(
            text: String,
            paint: TextPaint,
            spacingAfter: Float = 6f
        ) {
            if (
                text.isBlank()
            ) {
                return
            }

            val layout =
                StaticLayout
                    .Builder
                    .obtain(
                        text,
                        0,
                        text.length,
                        paint,
                        CONTENT_WIDTH
                    )
                    .setAlignment(
                        Layout.Alignment
                            .ALIGN_NORMAL
                    )
                    .setIncludePad(false)
                    .build()

            ensureSpace(
                layout.height
                    .toFloat() +
                    spacingAfter
            )

            val currentCanvas =
                canvas
                    ?: return

            currentCanvas.save()

            currentCanvas.translate(
                MARGIN,
                y
            )

            layout.draw(
                currentCanvas
            )

            currentCanvas.restore()

            y +=
                layout.height +
                    spacingAfter
        }

        private fun drawImageIfPossible(
            path: String
        ) {
            val file =
                File(path)

            if (
                !file.exists() ||
                !file.isFile
            ) {
                return
            }

            val bitmap =
                decodeBoundedBitmap(
                    file
                ) ?: return

            try {
                val maxWidth =
                    CONTENT_WIDTH
                        .toFloat()

                val maxHeight =
                    220f

                val scale =
                    minOf(
                        maxWidth /
                            bitmap.width,
                        maxHeight /
                            bitmap.height,
                        1f
                    )

                val width =
                    bitmap.width *
                        scale

                val height =
                    bitmap.height *
                        scale

                ensureSpace(
                    height +
                        12f
                )

                val destination =
                    RectF(
                        MARGIN,
                        y,
                        MARGIN +
                            width,
                        y +
                            height
                    )

                canvas?.drawBitmap(
                    bitmap,
                    null,
                    destination,
                    null
                )

                y +=
                    height +
                        12f

            } finally {
                bitmap.recycle()
            }
        }

        private fun decodeBoundedBitmap(
            file: File
        ): Bitmap? {

            val bounds =
                BitmapFactory.Options().apply {
                    inJustDecodeBounds =
                        true
                }

            BitmapFactory.decodeFile(
                file.absolutePath,
                bounds
            )

            if (
                bounds.outWidth <= 0 ||
                bounds.outHeight <= 0
            ) {
                return null
            }

            var sample =
                1

            while (
                bounds.outWidth /
                    sample >
                    1600 ||
                bounds.outHeight /
                    sample >
                    1600
            ) {
                sample *=
                    2
            }

            val options =
                BitmapFactory.Options().apply {
                    inSampleSize =
                        sample
                }

            return BitmapFactory.decodeFile(
                file.absolutePath,
                options
            )
        }

        private fun isLikelyVideo(
            path: String
        ): Boolean {
            val lower =
                path.lowercase()

            return lower.endsWith(
                ".mp4"
            ) ||
                lower.endsWith(
                    ".webm"
                ) ||
                lower.endsWith(
                    ".3gp"
                )
        }

        private fun ensurePage() {
            if (
                page == null
            ) {
                startPage()
            }
        }

        private fun ensureSpace(
            required: Float
        ) {
            ensurePage()

            if (
                y + required >
                PAGE_HEIGHT -
                    MARGIN
            ) {
                finishPage()
                startPage()
            }
        }

        private fun startPage() {
            pageNumber++

            val info =
                PdfDocument
                    .PageInfo
                    .Builder(
                        PAGE_WIDTH,
                        PAGE_HEIGHT,
                        pageNumber
                    )
                    .create()

            page =
                pdf.startPage(
                    info
                )

            canvas =
                page?.canvas

            y =
                MARGIN
        }

        private fun finishPage() {
            page?.let {
                current ->
                pdf.finishPage(
                    current
                )
            }

            page =
                null

            canvas =
                null
        }

        fun finish() {
            finishPage()
        }
    }
}

13.3 MIGRATION 4 -> 5 TEST
--------------------------

NEW FILE:
app/src/androidTest/java/com/onelineaday/dailydiary/data/JournalMigrationTest.kt

package com.onelineaday.dailydiary.data

import androidx.room.testing.MigrationTestHelper
import androidx.sqlite.db.framework.FrameworkSQLiteOpenHelperFactory
import androidx.test.ext.junit.runners.AndroidJUnit4
import androidx.test.platform.app.InstrumentationRegistry
import org.junit.Assert.assertEquals
import org.junit.Rule
import org.junit.Test
import org.junit.runner.RunWith

@RunWith(AndroidJUnit4::class)
class JournalMigrationTest {

    private val dbName =
        "migration-test"

    @get:Rule
    val helper =
        MigrationTestHelper(
            InstrumentationRegistry
                .getInstrumentation(),
            JournalDatabase::class.java,
            emptyList(),
            FrameworkSQLiteOpenHelperFactory()
        )

    @Test
    fun migration4To5PreservesEntryAndAddsAudioColumn() {

        helper
            .createDatabase(
                dbName,
                4
            )
            .apply {

                execSQL(
                    """
                    INSERT INTO journals(
                        id,
                        name,
                        colorHex,
                        createdAt
                    )
                    VALUES(
                        'default',
                        'My Diary',
                        NULL,
                        1
                    )
                    """.trimIndent()
                )

                execSQL(
                    """
                    INSERT INTO journal_entries(
                        date,
                        journalId,
                        content,
                        mood,
                        photoUri,
                        mediaUris,
                        tags,
                        isPinned,
                        createdAt,
                        updatedAt
                    )
                    VALUES(
                        '2026-09-21',
                        'default',
                        'migration memory',
                        'HAPPY',
                        NULL,
                        '[]',
                        '[]',
                        0,
                        1,
                        1
                    )
                    """.trimIndent()
                )

                close()
            }

        val migrated =
            helper.runMigrationsAndValidate(
                dbName,
                5,
                true,
                JournalDatabase
                    .MIGRATION_4_5
            )

        migrated
            .query(
                """
                SELECT
                    content,
                    audioUri,
                    journalId
                FROM journal_entries
                WHERE date = '2026-09-21'
                """.trimIndent()
            )
            .use {
                cursor ->

                require(
                    cursor.moveToFirst()
                )

                assertEquals(
                    "migration memory",
                    cursor.getString(
                        0
                    )
                )

                assertEquals(
                    null,
                    cursor.getString(
                        1
                    )
                )

                assertEquals(
                    "default",
                    cursor.getString(
                        2
                    )
                )
            }

        migrated.close()
    }
}

NOTE:
MigrationTestHelper constructor overloads can vary slightly between Room
versions. If Android Studio resolves the simpler constructor for Room 2.7.1,
use the IDE-provided 2.7.1 signature; do not change migration logic to satisfy
a constructor mismatch.
